    def __init__(self, id: int, nev: str, email: str = None):

        self.nev = nev
        self.email = email

    def __repr__(self):

A dataclass egy teljesen normális Python osztály, így ugyanúgy örökölhet, mint bármilyen más osztály.A mintádban az adat-objektum lesz az ősosztály, a leszármazott osztály pedig megörökli az összes adatmezőt, és te tetszőleges függvényeket (metódusokat) adhatsz hozzá.Itt van egy gyakorlati példa, hogyan néz ki ez a mentési logikában:Kód példa: Ősosztály mint adat, leszármazott metódusokkalpythonfrom dataclasses import dataclass

# 1. AZ ŐSOSZTÁLY: Csak az adatokat tárolja (Séma)
@dataclass
class MentesAdatBase:
    fajlnev: str
    meret_kb: int
    tipus: str

# 2. A LESZÁRMAZOTT: Megörökli az adatokat, és kap saját függvényeket
@dataclass
class MentendoFajl(MentesAdatBase):
    
    # Új függvény a leszármazottban
    def ellenoriz_meret(self) -> bool:
        if self.meret_kb > 10240:  # 10 MB limit
            print(f"Figyelem: A(z) {self.fajlnev} túl nagy!")
            return False
        return True

    # Egy másik függvény, ami elvégzi a mentést
    def ment(self):
        if self.ellenoriz_meret():
            print(f"[{self.tipus.upper()}] Sikeresen mentve: {self.fajlnev}")
Körültekintően használja a kódot.Hogyan működik ez a gyakorlatban?Amikor létrehozod a leszármazott objektumot, az ősosztály paramétereit kell neki átadnod:python# Létrehozzuk a leszármazottat az ősosztály adataival
doksi = MentendoFajl(fajlnev="riport.pdf", meret_kb=500, tipus="pdf")

# Elérjük a leszármazott saját függvényét
doksi.ment()
# Kiírás: [PDF] Sikeresen mentve: riport.pdf
